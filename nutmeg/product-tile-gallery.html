Product Tile Gallery

An edge-to-edge product grid with swipeable photos per tile and a filter and sort bar.

Live demo

Switch the screen width. Photos show only once they are on the image host.

Open in a new tab

How it's built

An edge-to-edge product grid with swipeable photos per tile and a filter and sort bar.

Notes from the code

  • Each tile: slide 1 is the product cut out from the site's image service (transparent PNG) with a soft drop shadow;
  • slide 2 is the plain back where it makes sense; then any cleared photos from the Tunnel.
  • A slide whose image fails removes itself, and the bars count what is left. One slide: no bars, no arrows.
  • Filter bar follows the Naadam study (19 Oct): quiet uppercase shortcuts, Sort on the right with its own small menu (black edge, notch, chosen order underlined, arrow keys and Escape work), one drawer, live results, removable chips, sticky bar, choices kept in the web address (pt_team, pt_player, pt_price, pt_sort). Set --campaign-sticky-top to the site header height in Fastr.
  • Filters work on the tiles in this block only; a full page version would read the site's own filters.
  • Grid: 2 columns, 3 from 640px, 4 from 1024px. 1px hairlines. The page grows with the grid, so no fixed height.

Fastr heights

No single set of heights.

This block's height changes with its settings or its content. Measure it for each use. The notes from the code say why.

Accessibility

    sortOpts.forEach(function(o){var on=o.getAttribute('data-value')===key;o.setAttribute('aria-selected',on?'true':'false');if(on)root.querySelector('.ptg-sort-val').textContent=o.getAttribute('data-short');});
    tiles.slice().sort(function(a,b){var pa=+a.getAttribute('data-price'),pb=+b.getAttribute('data-price');
      if(key==='low')return pa-pb;if(key==='high')return pb-pa;if(key==='saving')return +b.getAttribute('data-saving')-+a.getAttribute('data-saving');
      return +a.getAttribute('data-order')-+b.getAttribute('data-order');}).forEach(function(t){grid.appendChild(t);});
    Array.prototype.forEach.call(root.querySelectorAll('.ptg-group'),function(sec){
      var g=sec.getAttribute('data-group'),n=state[g].length;
      sec.querySelector('.ptg-n').textContent=n?' ('+n+')':'';
      var sh=root.querySelector('.ptg-short[data-open="'+g+'"] .ptg-n');if(sh)sh.textContent=n?' ('+n+')':'';
      Array.prototype.forEach.call(sec.querySelectorAll('input'),function(box){
        box.checked=state[g].indexOf(box.value)>-1;
        var c=tiles.filter(function(t){return match(t,g)&&val(t,g)===box.value;}).length;
        box.parentNode.querySelector('.ptg-c').textContent='('+c+')';
        box.disabled=!c&&!box.checked;
      });
    });
    var all=root.querySelector('.ptg-short[data-open="all"] .ptg-n');all.textContent=total()?' ('+total()+')':'';
    clear.hidden=!total();
    chips.innerHTML='';chips.hidden=!total();
    for(var g in state)state[g].forEach(function(v){var b=document.createElement('button');b.type='button';b.className='ptg-chip-x';b.setAttribute('aria-label','Remove filter: '+LABEL[v]);b.innerHTML=LABEL[v]+' <span aria-hidden="true">×</span>';b.addEventListener('click',(function(gg,vv){return function(){state[gg]=state[gg].filter(function(x){return x!==vv;});render();writeUrl();};})(g,v));chips.appendChild(b);});
    if(total()){var c=document.createElement('button');c.type='button';c.className='ptg-clearlink';c.textContent='Clear all';c.addEventListener('click',reset);chips.appendChild(c);}
  }
  function reset(){for(var g in state)state[g]=[];render();writeUrl();}
  function setGroup(sec,open){var b=sec.querySelector('.ptg-gh button');b.setAttribute('aria-expanded',open?'true':'false');sec.querySelector('.ptg-opts').hidden=!open;}
  function focusables(){return Array.prototype.filter.call(drawer.querySelectorAll('button,input,select'),function(e){return !e.disabled&&e.offsetParent!==null;});}
  function openDrawer(group,btn){
    opener=btn;drawer.hidden=false;scrim.hidden=false;document.documentElement.style.overflow='hidden';
    Array.prototype.forEach.call(root.querySelectorAll('.ptg-short'),function(s){s.setAttribute('aria-expanded',s===btn?'true':'false');});
    var target=group==='all'?'team':group;
    Array.prototype.forEach.call(drawer.querySelectorAll('.ptg-group'),function(sec){if(sec.getAttribute('data-group')===target)setGroup(sec,true);});
    var sec=drawer.querySelector('.ptg-group[data-group="'+target+'"]');var first=sec&&sec.querySelector('input:not(:disabled)');
    (first||drawer.querySelector('.ptg-close')).focus();
  }
  function closeDrawer(){drawer.hidden=true;scrim.hidden=true;document.documentElement.style.overflow='';
    Array.prototype.forEach.call(root.querySelectorAll('.ptg-short'),function(s){s.setAttribute('aria-expanded','false');});
    if(opener)opener.focus();}
  Array.prototype.forEach.call(root.querySelectorAll('.ptg-short'),function(b){b.addEventListener('click',function(){openDrawer(b.getAttribute('data-open'),b);});});
  Array.prototype.forEach.call(drawer.querySelectorAll('.ptg-group'),function(sec){
    var g=sec.getAttribute('data-group');
    sec.querySelector('.ptg-gh button').addEventListener('click',function(){setGroup(sec,this.getAttribute('aria-expanded')!=='true');});
    Array.prototype.forEach.call(sec.querySelectorAll('input'),function(box){box.addEventListener('change',function(){
      state[g]=state[g].filter(function(x){return x!==box.value;});if(box.checked)state[g].push(box.value);render();writeUrl();});});
  });
  drawer.querySelector('.ptg-close').addEventListener('click',closeDrawer);
  scrim.addEventListener('click',closeDrawer);view.addEventListener('click',closeDrawer);clear.addEventListener('click',reset);
  drawer.addEventListener('keydown',function(e){
    if(e.key==='Escape'){e.preventDefault();closeDrawer();return;}
    if(e.key!=='Tab')return;var f=focusables();if(!f.length)return;
    if(e.shiftKey&&document.activeElement===f[0]){e.preventDefault();f[f.length-1].focus();}
    else if(!e.shiftKey&&document.activeElement===f[f.length-1]){e.preventDefault();f[0].focus();}
  });
  function markActive(i){activeIx=(i+sortOpts.length)%sortOpts.length;sortOpts.forEach(function(o,k){o.classList.toggle('is-active',k===activeIx);});sortList.setAttribute('aria-activedescendant',sortOpts[activeIx].id);}
  function openSort(){sortList.hidden=false;sortBtn.setAttribute('aria-expanded','true');markActive(sortOpts.findIndex(function(o){return o.getAttribute('data-value')===sortKey;}));sortList.focus();}
  function closeSort(back){sortList.hidden=true;sortBtn.setAttribute('aria-expanded','false');if(back)sortBtn.focus();}
  function choose(i){sortKey=sortOpts[i].getAttribute('data-value');closeSort(true);render();writeUrl();}
  sortBtn.addEventListener('click',function(){sortList.hidden?openSort():closeSort(true);});
  sortBtn.addEventListener('keydown',function(e){if(e.key==='ArrowDown'||e.key==='ArrowUp'){e.preventDefault();openSort();}});
  sortOpts.forEach(function(o,i){o.addEventListener('click',function(){choose(i);});o.addEventListener('mousemove',function(){markActive(i);});});
  sortList.addEventListener('keydown',function(e){
    if(e.key==='ArrowDown'){e.preventDefault();markActive(activeIx+1);}
    else if(e.key==='ArrowUp'){e.preventDefault();markActive(activeIx-1);}
    else if(e.key==='Home'){e.preventDefault();markActive(0);}
    else if(e.key==='End'){e.preventDefault();markActive(sortOpts.length-1);}
    else if(e.key==='Enter'||e.key===' '){e.preventDefault();choose(activeIx);}
    else if(e.key==='Escape'){e.preventDefault();closeSort(true);}
    else if(e.key==='Tab'){closeSort(false);}
  });
  document.addEventListener('click',function(e){if(!sortList.hidden&&!root.querySelector('.ptg-sort').contains(e.target))closeSort(false);});
  var bar=root.querySelector('.ptg-bar'),sentinel=root.querySelector('.ptg-sentinel');
  if('IntersectionObserver' in window)new IntersectionObserver(function(es){bar.classList.toggle('is-stuck',!es[0].isIntersecting);}).observe(sentinel);
  readUrl();render();
  root.classList.add('is-ready');
})();
