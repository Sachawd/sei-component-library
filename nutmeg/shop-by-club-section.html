Shop by Club section

A photo band with a search box: type any club or player and the matching pages drop down inside the block.

Live demo

Switch the screen width. Photos show only once they are on the image host.

Open in a new tab

How it's built

A photo band with a search box: type any club or player and the matching pages drop down inside the block. The background photo is one line to set at the top of the style block.

Fastr heights

Type these into the three Reserved height boxes.

340pixels on phones (Mobile box).
380pixels from 640 wide.
412pixels from 1024 wide.

The code

Paste the whole file into one Fastr HTML component.

  var SRCH = "https://www.worldsoccershop.com/srch?query=";
  var BASE = "https://www.worldsoccershop.com/shop/clubs/";

  var root=document.getElementById("c17"), q=document.getElementById("c17q"),
      panel=document.getElementById("c17panel"), scroll=document.getElementById("c17scroll"),
      clearBtn=document.getElementById("c17x");
  if(!root||!q||!scroll) return;

  function fold(s){
    s=String(s||"").toLowerCase();
    try{ s=s.normalize("NFD").replace(/[̀-ͯ]/g,""); }catch(e){}
    return " "+s.replace(/[^a-z0-9]+/g," ").trim()+" ";
  }

  var CL=DATA.clubs||[], PL=DATA.players||[], TIFO=DATA.tifo||{};
  for(var i=0;i<CL.length;i++){
    var c=CL[i], hay=fold(c.n);
    for(var a=0;a<(c.a||[]).length;a++) hay+=fold(c.a[a]);
    c._h=hay; c._n=fold(c.n);
  }
  for(var p=0;p<PL.length;p++) PL[p]._h=fold(PL[p].n);

  function esc(s){return String(s).replace(/[&<>"']/g,function(m){
    return {"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[m];});}

  function matchedAlias(c,t){
    for(var a=0;a<(c.a||[]).length;a++){ if((" "+fold(c.a[a]).trim()).indexOf(t)===0) return c.a[a]; }
    for(var b=0;b<(c.a||[]).length;b++){ if(fold(c.a[b]).indexOf(t)>-1) return c.a[b]; }
    return null;
  }

  function search(raw){
    var t=fold(raw).trim(); if(!t) return {clubs:[],players:[]};
    t=" "+t;
    var starts=[],inside=[],ppl=[];
    for(var i=0;i<CL.length;i++){
      var c=CL[i];
      if(c._n.indexOf(t)===0) starts.push(c);
      else if(c._h.indexOf(t)>-1) inside.push(c);
    }
    for(var j=0;j<PL.length&&ppl.length<4;j++) if(PL[j]._h.indexOf(t)>-1) ppl.push(PL[j]);

    return {clubs:starts.concat(inside).slice(0,8), players:ppl, term:t};
  }

  function crest(c){
    if(!c.c) return '<span class="c17__crestph" aria-hidden="true"></span>';
    return '<img class="c17__crest" src="'+esc(c.c)+'" alt="" width="34" height="34" '+
           'onerror="this.onerror=null;this.style.visibility=\'hidden\';">';
  }
  function clubRow(c,note){
    return '<a class="c17__row" role="option" target="_top" href="'+esc(BASE+c.s)+'">'+
      crest(c)+'<span class="c17__t"><b>'+esc(c.n)+'</b><em>'+esc(note||c.l)+'</em></span>'+
      '<span class="c17__go">Shop &rarr;</span></a>';
  }
  function tifoRows(c){
    var arts=TIFO[c.n]; if(!arts) return "";
    var out="";
    for(var i=0;i<arts.length;i++){
      out+='<a class="c17__row c17__row--sub" role="option" target="_top" href="'+esc(arts[i][1])+'">'+
        '<span class="c17__doc" aria-hidden="true"></span>'+
        '<span class="c17__t"><b>'+esc(arts[i][0])+'</b><em>From The Tifo</em></span>'+
        '<span class="c17__go">Read &rarr;</span></a>';
    }
    return out;
  }

  function render(raw){
    var html="", r=search(raw);
    if(!raw.trim()){
      html='<div class="c17__kicker">Most shopped</div>';
      for(var i=0;i<8&&i<CL.length;i++) html+=clubRow(CL[i]);
    } else if(r.clubs.length||r.players.length){
      if(r.clubs.length){
        html+='<div class="c17__kicker">'+(r.clubs.length===1?"Club":"Clubs")+'</div>';
        for(var k=0;k<r.clubs.length;k++){
          var c=r.clubs[k], al=matchedAlias(c,r.term);
          html+=clubRow(c, al? c.l+" · matched “"+al+"”" : c.l);
          if(k===0) html+=tifoRows(c);
        }
      }
      if(r.players.length){
        html+='<div class="c17__kicker">Players</div>';
        for(var m=0;m<r.players.length;m++){
          var pl=r.players[m];
          html+='<a class="c17__row" role="option" target="_top" href="'+esc(pl.u)+'">'+
            '<span class="c17__crestph" aria-hidden="true"></span>'+
            '<span class="c17__t"><b>'+esc(pl.n)+'</b><em>'+esc(pl.c)+'</em></span>'+
            '<span class="c17__go">Shop &rarr;</span></a>';
        }
      }
    } else {
      html='<div class="c17__empty"><b>No club page for &ldquo;'+esc(raw.trim())+'&rdquo; yet.</b>'+
        '<span>We may still stock them &mdash; have a look.</span>'+
        '<a class="c17__cta" target="_top" href="'+esc(SRCH+encodeURIComponent(raw.trim()))+'">'+
        'Search the store &rarr;</a></div>'+
        '<div class="c17__kicker">Or jump to</div>';
      for(var z=0;z<4&&z<CL.length;z++) html+=clubRow(CL[z]);
    }
    scroll.innerHTML=html; sel=-1;
  }

  function fit(){
    var f=scroll.getBoundingClientRect(), b=root.getBoundingClientRect(), room=b.bottom-f.top-16;
    var h=Math.max(96,Math.min(340,room));
    scroll.style.maxHeight=h+"px";
  }
  function open(){
    if(root.classList.contains("is-open")){ fit(); return; }
    root.classList.add("is-open");
    q.setAttribute("aria-expanded","true");
    fit();
    requestAnimationFrame(fit);
    setTimeout(fit,200);
  }
  function close(){ root.classList.remove("is-open"); q.setAttribute("aria-expanded","false"); sel=-1; }

  var sel=-1;
  function rows(){ return scroll.querySelectorAll(".c17__row"); }
  function move(d){
    var r=rows(); if(!r.length)return;
    if(sel>-1&&r[sel]) r[sel].classList.remove("is-sel");
    sel=(sel+d+r.length)%r.length;
    r[sel].classList.add("is-sel");
    var el=r[sel], top=el.offsetTop, h=el.offsetHeight;
    if(top<scroll.scrollTop) scroll.scrollTop=top;
    else if(top+h>scroll.scrollTop+scroll.clientHeight) scroll.scrollTop=top+h-scroll.clientHeight;
  }

  q.addEventListener("input",function(){
    root.classList.toggle("is-typing", q.value.length>0);
    render(q.value); open();
  });
  q.addEventListener("focus",function(){ render(q.value); open(); });
  q.addEventListener("keydown",function(e){
    if(e.key==="ArrowDown"){e.preventDefault();move(1);}
    else if(e.key==="ArrowUp"){e.preventDefault();move(-1);}
    else if(e.key==="Enter"){ var r=rows(); if(sel>-1&&r[sel]){e.preventDefault();r[sel].click();} }
    else if(e.key==="Escape"){ close(); q.blur(); }
  });
  clearBtn.addEventListener("click",function(){
    q.value=""; root.classList.remove("is-typing"); render(""); q.focus();
  });
  document.addEventListener("click",function(e){
    if(!e.target.closest(".c17__searchwrap")) close();
  });
  window.addEventListener("resize",function(){ if(root.classList.contains("is-open")) fit(); });

  window.addEventListener("pageshow",function(e){ if(e.persisted) close(); });

  root.classList.add("is-ready");
})();
