Sale quick links strip

A row of round product pictures linking to sale categories.

Live demo

Switch the screen width. Photos show only once they are on the image host.

Open in a new tab

How it's built

A row of round product pictures linking to sale categories. Pictures come straight from the site's product image service.

Fastr heights

Type these into the three Reserved height boxes.

158pixels on phones (Mobile box).
122pixels from 640 wide.
122pixels from 1024 wide.

<script>
(function(){
  var root=document.getElementById('isel-sale-v1');
  if(!root)return;
  var list=root.querySelector('.isel-list');
  var steps=root.querySelectorAll('.isel-step');
  function update(){
    var over=list.scrollWidth>list.clientWidth+1;
    root.classList.toggle('is-scrollable',over);
    steps[0].disabled=list.scrollLeft<=1;
    steps[1].disabled=list.scrollLeft+list.clientWidth>=list.scrollWidth-1;
  }
  Array.prototype.forEach.call(steps,function(b){
    b.addEventListener('click',function(){
      var reduce=window.matchMedia('(prefers-reduced-motion: reduce)').matches;
      list.scrollBy({left:Number(b.getAttribute('data-dir'))*list.clientWidth*0.8,behavior:reduce?'auto':'smooth'});
    });
  });
  list.addEventListener('scroll',update,{passive:true});
  window.addEventListener('resize',update);
  if(document.fonts&&document.fonts.ready)document.fonts.ready.then(update);
  update();
})();
</script>
