<script>
/* The only script in this component, and it does two things, both driven by one
   piece of state: where are we in the scroll.

   1. EDGE FADE. A fade over the first product when there is nothing to its left
      does not read as "there is more this way", it reads as a rendering fault.
      That is what it looks like in Sacha's reference file and it is half of why
      he said it fades too much. Left edge is crisp until you scroll; right edge
      goes crisp at the end.
   2. ARROWS. Same state, so the fade and the arrows can never disagree about
      whether you are at an end. Added 2026-08-06 at Sacha's request, for
      desktop mouse users. They are display:none unless the pointer is fine, so
      a phone never renders them.

   Rule B2 is zero-JS by default; this is a documented exception. It touches
   nothing outside the iframe. With JS off you still get a scrollable rail with
   a visible scrollbar, a static fade and no arrows, which is the pre-08-06
   behaviour and is perfectly usable. */
(function () {
  var rail = document.getElementById('kr-rail');
  if (!rail) return;
  var prev = document.querySelector('[data-kr="prev"]');
  var next = document.querySelector('[data-kr="next"]');

  /* One card plus one gap, read from the DOM rather than hardcoded, so the
     three breakpoints do not each need their own number. Two cards per click
     is the live block's behaviour and it feels right at this card width. */
  function step() {
    var card = rail.querySelector('.kr__card');
    if (!card) return 300;
    var gap = parseFloat(getComputedStyle(rail).columnGap || getComputedStyle(rail).gap) || 20;
    return (card.getBoundingClientRect().width + gap) * 2;
  }
  function update() {
    var max = rail.scrollWidth - rail.clientWidth;
    var atStart = rail.scrollLeft <= 2;
    var atEnd = rail.scrollLeft >= max - 2;
    rail.classList.toggle('is--at-start', atStart);
    rail.classList.toggle('is--at-end', atEnd);
    /* If nothing overflows there is nothing to navigate, so BOTH go away.
       Without this you get two live-looking buttons that do nothing. */
    if (prev) prev.disabled = atStart || max <= 2;
    if (next) next.disabled = atEnd || max <= 2;
  }
  function go(dir) {
    var reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    rail.scrollBy({ left: dir * step(), behavior: reduce ? 'auto' : 'smooth' });
  }
  if (prev) prev.addEventListener('click', function () { go(-1); });
  if (next) next.addEventListener('click', function () { go(1); });

  var raf;
  rail.addEventListener('scroll', function () {
    cancelAnimationFrame(raf); raf = requestAnimationFrame(update);
  }, { passive: true });
  window.addEventListener('resize', update);
  update();
})();
</script>
