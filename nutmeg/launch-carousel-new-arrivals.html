(function () {

  function esc(s) {
    return String(s == null ? "" : s).replace(/[&<>"']/g, function (c) {

    });
  }

    return '<li class="lc__card">' +

      '<span class="lc__cat">' + esc(c.cat || "Jerseys") + '</span></span>' +
      '<span class="lc__label"><span class="club">' + esc(c.club) + '</span>' +
      (c.kit ? '<span class="bar">|</span><span class="kit">' + esc(c.kit) + '</span>' : '') +

  }

  var ctrl = window.AbortController ? new AbortController() : null;
  var timer = setTimeout(function () { if (ctrl) ctrl.abort(); }, 4000);
  fetch(FEED, { signal: ctrl ? ctrl.signal : undefined, credentials: "omit" })
    .then(function (r) { return r.ok ? r.json() : null; })
    .then(function (data) {
      clearTimeout(timer);
      var cards = data && Array.isArray(data.cards) ? data.cards.filter(function (c) {
        return c && c.club && c.month && c.day && IMG.test(c.image || "") && LINK.test(c.link || "");
      }) : [];
      if (!cards.length) { section.setAttribute("data-feed", "fallback"); return; }

      requestAnimationFrame(function () { rail.scrollTo({ left: 0, behavior: "instant" }); });
      section.setAttribute("data-feed", "live");
      if (data.generated) section.setAttribute("data-feed-generated", data.generated);
    })

})();

})();
