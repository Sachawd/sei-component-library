<script>
/* ═══════════════════════════════════════════════════════════════════════════
   SELECTION SECTION · tab engine + rail arrows
   ─────────────────────────────────────────────────────────────────────────
   Inline, self-contained, no CDN library, never touches the parent page.
   The whole thing is a progressive enhancement: it adds .is-ready, which is
   what reveals the tab strip. If this script never runs, panel 1 stands on
   its own and no dead control is shown.
   ═══════════════════════════════════════════════════════════════════════ */
(function () {
  /* Standalone preview: let the document scroll. Never fires inside Fastr. */
  try { if (window.self === window.top) document.documentElement.classList.add('is-standalone'); } catch (e) {}
  var root = document.getElementById('sel');
  if (!root) return;

  var tabs    = Array.prototype.slice.call(root.querySelectorAll('.sel-tab'));
  var panels  = Array.prototype.slice.call(root.querySelectorAll('.sel-panel'));
  var title   = document.getElementById('sel-head-title');
  var cta     = document.getElementById('sel-head-cta');
  var wrap    = document.getElementById('sel-railwrap');
  if (!tabs.length || tabs.length !== panels.length || !title || !cta || !wrap) return;

  var prevBtn = wrap.querySelector('[data-sel="prev"]');
  var nextBtn = wrap.querySelector('[data-sel="next"]');
  var active  = 0;

  function currentRail() {
    return panels[active].querySelector('.sel-rail');
  }

  /* Arrows exist only when the rail actually overflows. At five cards on a
     1392px container the row fills it exactly and there is nothing to page. */
  function syncOverflow() {
    var rail = currentRail();
    if (!rail) return;
    var over = rail.scrollWidth - rail.clientWidth > 2;
    wrap.classList.toggle('has-overflow', over);
    if (!over) return;
    var x = rail.scrollLeft;
    if (prevBtn) prevBtn.disabled = x <= 2;
    if (nextBtn) nextBtn.disabled = x >= (rail.scrollWidth - rail.clientWidth - 2);
  }

  function step(dir) {
    var rail = currentRail();
    if (!rail) return;
    var card = rail.querySelector('.sel-item');
    var by   = card ? (card.getBoundingClientRect().width + 8) : Math.round(rail.clientWidth * 0.8);
    rail.scrollBy({ left: dir * by, behavior: 'smooth' });
  }

  function select(i) {
    if (i === active) return;
    active = i;
    tabs.forEach(function (t, n) {
      t.setAttribute('aria-selected', n === i ? 'true' : 'false');
      t.setAttribute('tabindex', n === i ? '0' : '-1');
    });
    panels.forEach(function (p, n) {
      if (n === i) { p.removeAttribute('hidden'); } else { p.setAttribute('hidden', ''); }
    });

    /* The two things the designer annotation asks for: the head title becomes
       the tab's title, and the head CTA points at the tab's destination. The
       CTA LABEL is deliberately not changed. Figma gives one label only. */
    var t = tabs[i];
    title.textContent = t.getAttribute('data-title') || t.textContent;
    var href = t.getAttribute('data-cta');
    if (href) {
      cta.setAttribute('href', href);
      cta.setAttribute('aria-label', 'Shop all ' + title.textContent);
    }
    syncOverflow();
  }

  tabs.forEach(function (t, i) {
    t.setAttribute('tabindex', i === 0 ? '0' : '-1');
    t.addEventListener('click', function () { select(i); });
    t.addEventListener('keydown', function (e) {
      var n = null;
      if (e.key === 'ArrowRight') n = (i + 1) % tabs.length;
      if (e.key === 'ArrowLeft')  n = (i - 1 + tabs.length) % tabs.length;
      if (e.key === 'Home')       n = 0;
      if (e.key === 'End')        n = tabs.length - 1;
      if (n === null) return;
      e.preventDefault();
      select(n);
      tabs[n].focus();
    });
  });

  if (prevBtn) prevBtn.addEventListener('click', function () { step(-1); });
  if (nextBtn) nextBtn.addEventListener('click', function () { step(1); });

  panels.forEach(function (p) {
    var rail = p.querySelector('.sel-rail');
    if (rail) rail.addEventListener('scroll', syncOverflow, { passive: true });
  });

  var rt;
  window.addEventListener('resize', function () {
    clearTimeout(rt);
    rt = setTimeout(syncOverflow, 120);
  }, { passive: true });

  root.classList.add('is-ready');
  syncOverflow();

  /* Fonts land after first paint and can change a rail's scrollWidth, which
     changes whether the arrows belong on screen. Guarded: document.fonts is
     absent in older engines and its absence must not kill the tab engine. */
  if (document.fonts && document.fonts.ready && typeof document.fonts.ready.then === 'function') {
    document.fonts.ready.then(syncOverflow);
  }
})();
</script>
