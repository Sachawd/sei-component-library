(function(){
  var CONFIG = {
    assetBase: "https://www.imagehandler.net/dxassets/?img=Photos/",
    skyVimeoId: "1228462310",
    skyVimeoHash: "d7cf4e4369",
    stickers: ["gcdays-2026-10_sticker-goal-club-days.png", "gcdays-2026-10_sticker-25-off.png"],
    countdownStart: "2026-10-05T00:00:00-04:00",
    countdownEnd: "2026-10-08T23:59:59-04:00",
    beforeStartCopy: "Goal Club Days start Monday",
    runningCopy: "Goal Club Days left to order"
  };

  var root = document.getElementById("gcd-root");
  if (!root) return;
  var reduce = window.matchMedia && window.matchMedia("(prefers-reduced-motion: reduce)").matches;
  function $(sel){ return root.querySelector(sel); }
  function safe(fn){ try { fn(); } catch (e) { if (window.console) console.warn("Goal Club Days:", e); } }
  function load(src, ready, done){
    if (ready()) return done();
    var s = document.createElement("script");
    s.src = src; s.async = false; s.onload = done; s.onerror = done;
    document.head.appendChild(s);
  }
  function asset(file){ return CONFIG.assetBase && file ? CONFIG.assetBase + file : ""; }

  safe(function(){
    var box = $("[data-gcd-count]");
    var end = CONFIG.countdownEnd ? new Date(CONFIG.countdownEnd).getTime() : NaN;
    var start = CONFIG.countdownStart ? new Date(CONFIG.countdownStart).getTime() : NaN;
    if (!box || isNaN(end)) return;
    var row = $("[data-gcd-row]"), copy = $("[data-gcd-copy]");
    var dEl = $("[data-gcd-days]"), hEl = $("[data-gcd-hours]"), mEl = $("[data-gcd-mins]");
    function tick(){
      var now = Date.now(), left = end - now;
      if (left <= 0) { box.hidden = true; return false; }
      if (!isNaN(start) && now < start) {
        row.hidden = true;
        copy.textContent = CONFIG.beforeStartCopy;
      } else {
        var mins = Math.floor(left / 60000);
        dEl.textContent = Math.floor(mins / 1440);
        hEl.textContent = Math.floor((mins % 1440) / 60);
        mEl.textContent = mins % 60;
        row.hidden = false;
        copy.textContent = CONFIG.runningCopy;
      }
      box.hidden = false;
      return true;
    }
    if (tick()) var timer = setInterval(function(){ if (!tick()) clearInterval(timer); }, 30000);
  });

  if (reduce) return;

  safe(function(){
    var box = $("[data-gcd-sky]");
    if (!box || !CONFIG.skyVimeoId) return;
    var f = document.createElement("iframe");
    f.src = "https://player.vimeo.com/video/" + CONFIG.skyVimeoId + "?" + (CONFIG.skyVimeoHash ? "h=" + CONFIG.skyVimeoHash + "&" : "") +
      "background=1&autoplay=1&muted=1&loop=1&autopause=0&dnt=1";
    f.title = "";
    f.setAttribute("allow", "autoplay; fullscreen; picture-in-picture");
    f.setAttribute("referrerpolicy", "strict-origin-when-cross-origin");
    f.setAttribute("tabindex", "-1");
    f.setAttribute("aria-hidden", "true");
    window.addEventListener("message", function onReady(e){
      if (e.source !== f.contentWindow) return;
      var data = e.data;
      if (typeof data === "string") { try { data = JSON.parse(data); } catch (err) { return; } }
      if (!data || data.event !== "ready") return;
      window.removeEventListener("message", onReady);
      setTimeout(function(){ f.classList.add("is-ready"); }, 700);
    });
    box.appendChild(f);
  });

  // Osmo · Parallax Image Layers
  // ⚠️DEVIATION 1: scrub 0 → 0.6 so layers ease behind the scroll, off under reduced motion
  // ⚠️DEVIATION 2: scoped to this block's root, the host page is never queried
  function initParallaxLayers(){
    root.querySelectorAll("[data-parallax-layers]").forEach(function(triggerElement){
      var tl = gsap.timeline({scrollTrigger:{trigger:triggerElement, start:"0% 0%", end:"100% 0%", scrub:0.6}});
      var layers = [{layer:"1", yPercent:70},{layer:"2", yPercent:55},{layer:"3", yPercent:40},{layer:"4", yPercent:10}];
      layers.forEach(function(layerObj, idx){
        tl.to(triggerElement.querySelectorAll('[data-parallax-layer="' + layerObj.layer + '"]'), {yPercent:layerObj.yPercent, ease:"none"}, idx === 0 ? undefined : "<");
      });
    });
  }

  // Osmo · Falling 2D Objects (MatterJS)
  // ⚠️DEVIATION 4: target id is gcd-canvas-target; Fastr has no sandbox, so the id must be unique on the host page
  // ⚠️DEVIATION 5: textures come from CONFIG.stickers; the stickers are 366x392 with a 336px disc and a drop shadow, so the sprite scales and centres on the disc, not on a 256px square
  // ⚠️DEVIATION 6: object size is clamped to 72–170px so it reads on a phone and does not crowd a wide hero
  // ⚠️DEVIATION 7: pixelRatio follows the device, capped at 2
  function initFalling2DMatterJS(textures){

    const canvas = document.querySelector("#gcd-canvas-target");

    if (!canvas) {
      return;
    }

    const canvasWidth = canvas.clientWidth + 2;
    const canvasHeight = canvas.clientHeight + 2;
    const canvasWallDepth = canvasWidth / 4;
    const smileyAmount = 15;
    const smileySize = Math.max(72, Math.min(170, canvasWidth / 7.5));
    const smileySizeTexture = 336;
    const smileySizeScale = smileySize / smileySizeTexture;
    const smileyRestitution = 0.75;
    const worldGravity = 2;

    let { Engine, Render, Runner, Bodies, Composite, Mouse, MouseConstraint } = Matter;

    let engine = Engine.create();
    engine.world.gravity.y = worldGravity;

    let render = Render.create({
      element: canvas,
      engine: engine,
      options: {
        background: "transparent",
        wireframes: false,
        width: canvasWidth,
        height: canvasHeight,
        pixelRatio: Math.min(window.devicePixelRatio || 1, 2),
        border: "none",
      }
    });

    function getRandomNumber(min, max) {
      return Math.random() * (max - min) + min;
    }

    let min = smileySize / 2;
    let max = canvasWidth - (smileySize / 2);

    let textureIndex = 0;
    function getNextTexture() {
      const texture = textures[textureIndex];
      textureIndex = (textureIndex + 1) % textures.length;
      return texture;
    }

    const smileyCreate = () => {
      let smiley = Bodies.rectangle(getRandomNumber(min, max), smileySize, smileySize,
        smileySize, {
          chamfer: {
            radius: smileySize / 2
          },
          restitution: smileyRestitution,
          render: {
            sprite: {
              texture: getNextTexture(),
              xScale: smileySizeScale,
              yScale: smileySizeScale,
              yOffset: -0.064
            }
          }
        });
      Composite.add(engine.world, smiley);
    };

    let boxTop = Bodies.rectangle(canvasWidth / 2 + (canvasWallDepth * 2), canvasHeight +
      canvasWallDepth, canvasWidth + (canvasWallDepth * 4), (canvasWallDepth * 2), {
        isStatic: true
      });

    let boxLeft = Bodies.rectangle((canvasWallDepth * -1), canvasHeight / 2, (canvasWallDepth * 2),
      canvasHeight, {
        isStatic: true
      });

    let boxRight = Bodies.rectangle(canvasWidth + canvasWallDepth, canvasHeight / 2, (
      canvasWallDepth * 2), canvasHeight, {
      isStatic: true
    });

    let boxBottom = Bodies.rectangle(canvasWidth / 2 + (canvasWallDepth * 2), (canvasWallDepth * -1),
      canvasWidth + (canvasWallDepth * 4), (canvasWallDepth * 2), {
        isStatic: true
      });

    Composite.add(engine.world, [boxTop, boxLeft, boxRight, boxBottom]);

    Render.run(render);

    let runner = Runner.create();

    Matter.Runner.run(runner, engine);

    function repeatedFunction(count, maxCount) {
      if (count < maxCount) {
        smileyCreate();
        setTimeout(() => {
          repeatedFunction(count + 1, maxCount);
        }, 100);
      }
    }
    repeatedFunction(0, smileyAmount);

    let mouse = Mouse.create(render.canvas);
    let mouseConstraint = MouseConstraint.create(engine, {
      mouse: mouse,
      constraint: {
        stiffness: 0.2,
        render: {
          visible: false
        }
      }
    });

    Composite.add(engine.world, mouseConstraint);

    mouseConstraint.mouse.element.removeEventListener("mousewheel", mouseConstraint.mouse.mousewheel);
    mouseConstraint.mouse.element.removeEventListener("DOMMouseScroll", mouseConstraint.mouse.mousewheel);

    mouseConstraint.mouse.element.removeEventListener('touchstart', mouseConstraint.mouse.mousedown);
    mouseConstraint.mouse.element.removeEventListener('touchmove', mouseConstraint.mouse.mousemove);
    mouseConstraint.mouse.element.removeEventListener('touchend', mouseConstraint.mouse.mouseup);

    mouseConstraint.mouse.element.addEventListener('touchstart', mouseConstraint.mouse.mousedown, { passive: true });
    mouseConstraint.mouse.element.addEventListener('touchmove', (e) => {
      if (mouseConstraint.body) {
        mouseConstraint.mouse.mousemove(e);
      }
    });
    mouseConstraint.mouse.element.addEventListener('touchend', (e) => {
      if (mouseConstraint.body) {
        mouseConstraint.mouse.mouseup(e);
      }
    });
  }

  load("https://cdn.jsdelivr.net/npm/gsap@3.15/dist/gsap.min.js", function(){ return typeof window.gsap !== "undefined"; }, function(){
    if (typeof window.gsap === "undefined") return;
    load("https://cdn.jsdelivr.net/npm/gsap@3.15/dist/ScrollTrigger.min.js", function(){ return typeof window.ScrollTrigger !== "undefined"; }, function(){
      if (typeof window.ScrollTrigger === "undefined") return;
      gsap.registerPlugin(ScrollTrigger);
      safe(initParallaxLayers);
    });
  });

  var stickerUrls = CONFIG.stickers.map(asset).filter(Boolean);
  if (!stickerUrls.length) return;
  var textures = [], pending = stickerUrls.length;
  function stickerReady(){
    pending -= 1;
    if (pending > 0 || !textures.length) return;
    load("https://cdnjs.cloudflare.com/ajax/libs/matter-js/0.19.0/matter.min.js", function(){ return typeof window.Matter !== "undefined"; }, function(){
      if (typeof window.Matter === "undefined") return;
      safe(function(){ initFalling2DMatterJS(textures); });
    });
  }
  stickerUrls.forEach(function(src){
    var img = new Image();
    img.onload = function(){ textures.push(src); stickerReady(); };
    img.onerror = stickerReady;
    img.src = src;
  });
})();
