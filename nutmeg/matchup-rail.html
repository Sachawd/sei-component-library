  DOCTYPE IS THE FIRST BYTE ON PURPOSE (rules B3).

  ── WHAT IT IS ────────────────────────────────────────────────────────────
  Eight league tabs. One rail of match cards per tab. Native overflow-x
  scroll, hairline arrows on desktop only, no dots (canon: a free-scroll rail
  cannot honestly report a page). Team names link to their WSS club landing
  page when one exists; when one does not, the name is plain text.

  Three card states, three chip grounds, and the middle one is a deliberate
  call:
    upcoming    kickoff time in the VIEWER'S timezone, quiet #F0EBE6 chip
    live        reads "IN PROGRESS", inverted chip, NO SCORE IS SHOWN
    final       reads "FT", YELLOW chip, score shown
  A 15-minute-stale live score is wrong and readers notice. A blank one is
  not. This keeps the block honest at a refresh cadence that is not live.

  NO WIN/LOSS STYLING, on purpose. No green, no bold winner, no W/L badge.
  Both teams carry the same weight. ESPN and Fox colour the winner because
  they are sports media. This is a shop, and half of every audience just lost.

  ── DATA ──────────────────────────────────────────────────────────────────
  The block reads ONE JSON payload. It never calls a paid API: the key would
  be in view-source. Shape is normalised by the sync, not raw provider JSON,
  so swapping providers never touches this file.

    DATA_URL = null   → uses the inline SAMPLE_DATA (today, design review)
    DATA_URL = "..."  → fetches that URL, falls back to SAMPLE_DATA on failure

  Planned live path (not built): scheduled sync calls the provider for the
  eight leagues every ~15 min, writes one small JSON, this block reads it.
  Call volume is then fixed regardless of site traffic, which is the whole
  point of the arrangement.

  Payload contract:
    { "generated": ISO8601,
      "leagues": [ { "id", "name", "shopUrl", "matches": [
        { "kickoff": ISO8601 (UTC),
          "status": "upcoming" | "live" | "final",
          "home": { "name", "short"?, "url": string|null, "crest"?: string },
          "away": { "name", "short"?, "url": string|null, "crest"?: string },
          "score": { "home": int, "away": int } | null } ],
        "next": ISO8601 (UTC) | null } ] }

  next is optional: the league's earliest upcoming kickoff, looked up to 14
  days ahead, so a tab with no cards can still say when play resumes.

  short is optional and DISPLAY ONLY. Long names do not fit beside a crest and
  a score, so the sample carries: Brighton, Wolves, Nottm Forest, PSG,
  Marseille, Lyon. aria-label always uses the full name.
  crest is optional. Omit it and the club renders as initials.

  url:null is REQUIRED for any club with no landing page. The renderer prints
  those as plain text. It never constructs a /shop/clubs/ URL from a name
  (THE URL RULE, Sacha 08-03). Six clubs in the sample are deliberately
  unmapped so the fallback is visible in review: Nottingham Forest, Brentford,
  Leeds United, Ipswich Town, Wolverhampton Wanderers, AS Monaco. Wolves is
  the useful one to look at: it has a crest but no landing page, so it shows
  the badge and plain, unlinked text.

  /* ── formatting ─────────────────────────────────────────────────────── */
  function fmtDate(iso){
    var d = new Date(iso);
    if(isNaN(d)) return '';
    try{
      /* comma stripped: the meta line is uppercased, and "SUN, AUG 23" reads
         worse than "SUN AUG 23". Order still follows the viewer's locale. */
      return new Intl.DateTimeFormat(undefined,{weekday:'short',day:'numeric',month:'short'})
        .format(d).replace(/,/g,'');
    }catch(e){
      return d.toDateString().slice(0,10);
    }
  }
  function fmtTime(iso){
    var d = new Date(iso);
    if(isNaN(d)) return '';
    try{
      return new Intl.DateTimeFormat(undefined,{hour:'numeric',minute:'2-digit'}).format(d);
    }catch(e){
      return ('0'+d.getHours()).slice(-2)+':'+('0'+d.getMinutes()).slice(-2);
    }
  }

  /* "Sat 3 Oct", in the viewer's timezone. Day-before-month is Sacha's
     ruling for this line (24 Sept), so the locale is fixed here. */
  function fmtNext(iso){
    var d = new Date(iso);
    if(isNaN(d)) return '';
    try{
      return new Intl.DateTimeFormat('en-GB',{weekday:'short',day:'numeric',month:'short'})
        .format(d).replace(/,/g,'');
    }catch(e){
      var p = d.toDateString().split(' ');
      return p[0] + ' ' + Number(p[2]) + ' ' + p[1];
    }
  }

    return (d && !isNaN(d) && d.getTime() > Date.now()) ? iso : null;
  }

  /* Two letters, used when a club has no crest. Never a guessed image. */
  function initialsOf(name){
    var parts = String(name || '').replace(/[^A-Za-z0-9 ]/g,' ').split(/\s+/).filter(Boolean);
    if(!parts.length) return '';
    if(parts.length === 1) return parts[0].slice(0,2).toUpperCase();
    return (parts[0].charAt(0) + parts[1].charAt(0)).toUpperCase();
  }

    var slot = document.createElement('span');
    slot.className = 'mr-crest';
    var ini = document.createElement('span');
    ini.className = 'mr-crest__initials';
    ini.textContent = initialsOf(team && team.name);
    slot.appendChild(ini);
    var url = (team && typeof team.crest === 'string' && team.crest) ? team.crest : null;
    if(url){
      var img = document.createElement('img');
      img.className = 'mr-crest__img';
      img.alt = '';
      img.loading = 'lazy';
      img.decoding = 'async';
      /* exactly ONE self-disarming onerror, never a loop (rule D-images).
         On failure the crest fades and the initials underneath show. */
      img.onerror = function(){ this.onerror = null; this.style.opacity = '0'; };
      img.src = url;
      slot.appendChild(img);
    }
    return slot;
  }
  function teamEl(team){

  }

    var card = document.createElement('article');

    card.setAttribute('role','listitem');

    var rows = document.createElement('div');
    rows.className = 'mr-card__rows';
    rows.appendChild(rowEl(m.home, showScore ? m.score.home : null));
    rows.appendChild(rowEl(m.away, showScore ? m.score.away : null));
    card.appendChild(rows);

    foot.className = 'mr-card__foot';
    var chip = document.createElement('span');
    chip.className = 'mr-status';
    if(m.status === 'final')      chip.textContent = 'FT';
    else if(m.status === 'live')  chip.textContent = 'In progress';
    else                          chip.textContent = fmtTime(m.kickoff);
    foot.appendChild(chip);
    card.appendChild(foot);

    return card;
  }

  function renderRail(){
    var league = DATA.leagues[active];
    rail.textContent = '';
    if(!league) return;

    if(cta && league.shopUrl){
      cta.href = league.shopUrl;
      var ctaLong = document.getElementById('mr-cta-long');
      if(ctaLong) ctaLong.textContent = 'Shop ' + league.name;
      cta.setAttribute('aria-label', 'Shop ' + league.name);
    }

    var list = (league.matches || []).slice().sort(function(a,b){
      return new Date(a.kickoff) - new Date(b.kickoff);
    });

    if(!list.length){
      rail.appendChild(emptyEl(league));
    } else {
      var frag = document.createDocumentFragment();
      for(var i=0;i<list.length;i++) frag.appendChild(cardEl(list[i]));
      rail.appendChild(frag);
    }
    rail.scrollLeft = 0;
    sync();
  }

  function renderTabs(){
    tabs.textContent = '';
    DATA.leagues.forEach(function(lg, i){
      var b = document.createElement('button');
      b.className = 'mr-tab';
      b.type = 'button';
      b.setAttribute('role','tab');
      b.setAttribute('aria-selected', i === active ? 'true' : 'false');
      var lab = document.createElement('span');
      lab.className = 'mr-tab__label';
      lab.textContent = lg.name;
      var rule = document.createElement('span');
      rule.className = 'mr-tab__rule';
      b.appendChild(lab); b.appendChild(rule);
      b.addEventListener('click', function(){
        if(i === active) return;
        active = i;
        renderTabs();
        renderRail();
      });
      tabs.appendChild(b);
    });
  }

  /* ── arrows ─────────────────────────────────────────────────────────── */
  function sync(){
    var over = rail.scrollWidth - rail.clientWidth > 2;
    root.classList.toggle('has-overflow', over);
    if(!over){
      if(prev) prev.disabled = true;
      if(next) next.disabled = true;
      return;
    }
    var x = rail.scrollLeft;
    if(prev) prev.disabled = x <= 2;
    if(next) next.disabled = x >= (rail.scrollWidth - rail.clientWidth - 2);
  }
  function step(d){
    var c = rail.querySelector('.mr-card');
    var by = c ? (c.getBoundingClientRect().width + 8) : Math.round(rail.clientWidth * 0.8);
    rail.scrollBy({left: d * by, behavior:'smooth'});
  }
  if(prev) prev.addEventListener('click', function(){ step(-1); });
  if(next) next.addEventListener('click', function(){ step(1); });
  rail.addEventListener('scroll', sync, {passive:true});
  var rt;
  window.addEventListener('resize', function(){ clearTimeout(rt); rt = setTimeout(sync,120); }, {passive:true});

  /* ── boot ───────────────────────────────────────────────────────────── */
  /* Open on the first tab, in tab order, that has a match to show. */
  function firstLive(){
    for(var i=0;i<DATA.leagues.length;i++){
      var m = DATA.leagues[i] && DATA.leagues[i].matches;
      if(m && m.length) return i;
    }
    return 0;
  }

  function boot(){
    active = firstLive();
    renderTabs();
    renderRail();
    root.classList.add('is-ready');
    /* bring an off-screen default tab into view. scrollLeft, not
       scrollIntoView, which can scroll the parent page. */
    var sel = tabs.children[active];
    if(sel && active > 0){
      var over = sel.getBoundingClientRect().right - tabs.getBoundingClientRect().right;
      if(over > 0) tabs.scrollLeft += over + 16;
    }
    if(document.fonts && document.fonts.ready && typeof document.fonts.ready.then === 'function'){
      document.fonts.ready.then(sync);
    }
  }

  if(DATA_URL){
    /* Live path. A failed fetch keeps the last shipped payload rather than
       showing an empty band. Never blocks first paint for long: same-origin
       CDN JSON, and boot() runs either way. */
    var done = false;
    var finish = function(){ if(!done){ done = true; boot(); } };
    setTimeout(finish, 4000);
    try{
      fetch(DATA_URL, {cache:'no-store'})
        .then(function(r){ return r.ok ? r.json() : null; })
        .then(function(j){ if(j && j.leagues && j.leagues.length) DATA = j; })
        .catch(function(){})
        .then(finish);
    }catch(e){ finish(); }
  } else {
    boot();
  }
})();
