How it's built

Twelve tall player photos on a swipe rail, title and Shop all on top, arrows on desktop. Replaces the old Shop by Player section.

Fastr heights

Type these into the three Reserved height boxes.

568pixels from 1024 wide.

(function(){
  var root=document.getElementById('sb-players-v1'); if(!root) return;
  var rail=root.querySelector('.sb-rail'); if(!rail) return;
  var prev=root.querySelector('[data-sb="prev"]'), next=root.querySelector('[data-sb="next"]');
  function sync(){
    root.style.setProperty('--sb-rail-top',rail.offsetTop+'px');
    var over=rail.scrollWidth-rail.clientWidth>2;
    root.classList.toggle('has-overflow',over);
    if(!over) return;
    var x=rail.scrollLeft;
    if(prev) prev.disabled=x<=2;
    if(next) next.disabled=x>=(rail.scrollWidth-rail.clientWidth-2);
  }
  function step(d){
    var t=rail.querySelector('.sb-tile');
    var by=t?(t.getBoundingClientRect().width+8):Math.round(rail.clientWidth*0.8);
    rail.scrollBy({left:d*by,behavior:'smooth'});
  }
  if(prev) prev.addEventListener('click',function(){step(-1)});
  if(next) next.addEventListener('click',function(){step(1)});
  rail.addEventListener('scroll',sync,{passive:true});
  var rt; window.addEventListener('resize',function(){clearTimeout(rt);rt=setTimeout(sync,120)},{passive:true});
  root.classList.add('is-ready'); sync();
  if(document.fonts&&document.fonts.ready&&typeof document.fonts.ready.then==='function'){document.fonts.ready.then(sync);}
})();
</script>
