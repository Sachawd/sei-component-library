(function () {
  var roots = document.querySelectorAll('.stl[data-stl]:not(.stl--js)');
  Array.prototype.forEach.call(roots, function (root) {
    var viewport = root.querySelector('[data-stl-viewport]');
    var rail = root.querySelector('[data-stl-rail]');

    var closeBtn = root.querySelector('[data-stl-close]');
    var prev = root.querySelector('[data-stl-prev]');
    var next = root.querySelector('[data-stl-next]');
    if (!viewport || !rail || !drawer || !list) return;

    var cards = Array.prototype.slice.call(rail.querySelectorAll('[data-stl-card]'));
    var reduce = window.matchMedia('(prefers-reduced-motion: reduce)');
    var wide = window.matchMedia('(min-width: 768px)');
    var openItem = null;
    var opener = null;
    var ticking = false;

    function light(item, on) { item.classList.toggle('is-lit', on); }

    root.classList.add('stl--js');
    Array.prototype.forEach.call(rail.querySelectorAll('[data-stl-pic]'), function (pic) {
      pic.setAttribute('tabindex', '-1');

    });

    function px(el, prop) { return parseFloat(getComputedStyle(el)[prop]) || 0; }

    function closeChip() {
      if (!openItem) return;
      openItem.classList.remove('is-open');
      openItem.querySelector('[data-stl-dot]').setAttribute('aria-expanded', 'false');
      openItem.querySelector('[data-stl-chip]').hidden = true;
      openItem = null;
    }

    function toggleChip(item) {
      var same = item === openItem;
      closeChip();
      if (same) return;
      item.classList.add('is-open');
      item.querySelector('[data-stl-dot]').setAttribute('aria-expanded', 'true');
      item.querySelector('[data-stl-chip]').hidden = false;
      openItem = item;

    function fillDrawer(card) {
      var club = card.getAttribute('data-stl-club') || '';
      var all = card.querySelector('[data-stl-all]');
      drawerTitle.textContent = club;
      if (drawerAll && all) {
        drawerAll.setAttribute('href', all.getAttribute('href'));
        drawerAll.querySelector('.button-009__text').textContent = 'Shop ' + club;
