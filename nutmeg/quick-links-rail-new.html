  descriptive. THIS BLOCK IS PUBLIC. Putting an embargoed kit on a PLP leaks it.

  --ql-gap:16px;                   /* between chips. 8px under 640. */
  --ql-chip-h:70px;                /* THE height budget lives here. Figma's
                                      4 + 62 + 4. Still 70 after DEVIATION-3;
                                      the media just fills it now. */
  --ql-media:70px;                 /* was 62 inset, now flush. Tracks --ql-chip-h:
                                      change one, change both, or the media stops
                                      being square. */
  --ql-label-w:126px;
  --ql-gutter:52px;                /* the arrow's lane, Figma */
  --ql-arrow:48px;
  --ql-arrow-icon:24px;

}

html,body{

  overflow-x:hidden;
  overflow-y:hidden;               /* let a vertical wheel reach the parent page */
  touch-action:pan-y;              /* rules B4. The vertical axis belongs to the page. */
}

.ql{
  font-family:var(--ql-font);
  color:var(--ql-ink);
  -webkit-font-smoothing:antialiased;
  background:var(--ql-paper);
  position:relative;
  width:100%;
  padding:var(--ql-pad-y) 0;
}

/* ── THE RAIL ─────────────────────────────────────────────────────────────
   Native scroll, no engine. A quick-links row is 15 short items, not a 30-tile
   infinite grid, and GSAP here would be weight for nothing. Arrows are added on
   top for desktop mouse users, because a mouse wheel cannot scroll a horizontal
   rail (design canon). No dots: a free-scroll rail cannot honestly report a
   "page". ── */
.ql__rail{
  display:flex;
  align-items:center;
  gap:var(--ql-gap);
  list-style:none;
  margin:0;
  padding:0 var(--ql-gutter) 0 0;
  overflow-x:auto;
  overflow-y:hidden;
  scroll-snap-type:x proximity;
  overscroll-behavior-x:contain;
  touch-action:pan-x pinch-zoom;
  -webkit-overflow-scrolling:touch;
  scrollbar-width:none;            /* the arrows and the overflow are the affordance */
}
.ql__rail::-webkit-scrollbar{display:none}

/* ── "JUMP TO" ────────────────────────────────────────────────────────────
   First child of the rail, so it scrolls with the chips. That is what Figma
   does. Its left half is the 52px transparent lane the left arrow sits in.
   PINNED-LABEL: to pin it instead, change this one line to
       position:sticky;left:0;z-index:2;background:var(--ql-paper);
   and nothing else needs to move. ── */
.ql__jump{
  flex:0 0 auto;
  scroll-snap-align:start;         /* ⚠️ LOAD-BEARING. Without it the rail lands at
                                      scrollLeft 138 on load and "Jump To" is already
                                      scrolled off before anyone touches it: the label
                                      was not a snap target, so the browser snapped
                                      past it to the first chip. Measured, not
                                      theorised. Same class of bug as C-16's
                                      scroll-padding-inline note. */
  display:flex;
  align-items:center;
  height:var(--ql-chip-h);
  padding-left:var(--ql-gutter);   /* the arrow's lane */
  padding-right:8px;
  font-size:14px;
  font-weight:700;
  line-height:20px;
  letter-spacing:0.07px;
  color:var(--ql-label-ink);
  white-space:nowrap;
  -webkit-user-select:none;user-select:none;
}

/* ── THE CHIP ─────────────────────────────────────────────────────────────
   Fixed 70px height on BOTH variants, image and text-only, so the row stays
   level and the iframe budget holds no matter which mix a location needs. ── */
.ql__item{flex:0 0 auto;scroll-snap-align:start}
.ql__chip{
  display:flex;
  align-items:center;
  justify-content:center;
  /* ⚠️ gap:0, NOT Figma's 8px, and this is the second half of DEVIATION-3.
     Sacha, 08-18: "these are ALMOST filling but not quite... if you look at the
     logos they aren't perfectly centred."
     Both symptoms were the same 8px flex gap. It sat between the MEDIA and the
     DIVIDER, so a render with a baked #f2f2f2 ground stopped 8px short of the
     rule and left a white strip that read as "the image failed to fill". And it
     made the crests look shifted left: the eye reads media+gap (78px) as the
     picture area and centres against that, while the crest is centred in its
     own 70px. Off by exactly half the gap, 4px, which is precisely how far out
     they look.
     The 8px now lives on .ql__rule as margin-right, so the media butts the
     divider and the divider keeps its breathing room before the text. */
  gap:0;
  height:var(--ql-chip-h);
  box-sizing:border-box;
  /* ⚠️ DEVIATION-3 (Sacha, 08-18, after seeing it live): the media BLEEDS to the
     chip edge instead of sitting in Figma's 4px/8px inset. His words: "can we
     have the image field fill vs have padding".
     Right call. At 62px inset in a white chip on a white page the picture read as
     a stamp floating in space; flush, the chip reads as one object with a picture
     side and a text side, and the vertical rule finally has a job.
     Chip padding drops to right-only. The media grows 62 -> 70 to fill the full
     chip height, and because the old geometry was 8px pad + 62px media = 70px,
     the chip's overall width does not move. Same budget, bigger picture.
     overflow:hidden is what clips the media's left corners to the 4px radius. */
  padding:0 8px 0 0;
  overflow:hidden;
  background:var(--ql-paper);
  border:1px solid var(--ql-chip-line);
  border-radius:4px;
  text-decoration:none;
  color:var(--ql-ink);
  -webkit-tap-highlight-color:transparent;
  transition:border-color .25s var(--ql-ease), background-color .25s var(--ql-ease);
}
.ql__chip:focus-visible{outline:2px solid var(--ql-ink);outline-offset:2px}
@media (hover:hover) and (pointer:fine){
  .ql__chip:hover{border-color:var(--ql-ink)}
}
@media (prefers-reduced-motion:reduce){
  .ql__chip{transition:none}
}

.ql__media{
  flex:0 0 auto;
  display:block;
  width:var(--ql-media);
  height:var(--ql-media);   /* = --ql-chip-h, so it fills top to bottom */
  overflow:hidden;
}
.ql__img{
  display:block;width:100%;height:100%;
  object-fit:contain;              /* crests are logo-shaped, product renders are
                                      square. contain is the only thing that is
                                      right for both. Nothing is cropped. */
}

/* ── PHOTO CHIPS ──────────────────────────────────────────────────────────
   For images coming out of the S3 asset library, which is 3:4 / 16:9 / 4:5
   campaign photography rather than square cutouts. contain would letterbox a
   16:9 frame into a 62x35 sliver; cover centre-crops it to fill the square.
   The tile behind it stops a 1px rounding gap showing as a white seam.
   Crests must NEVER carry this class: cover would clip the badge. ── */
.ql__chip.is--photo .ql__media{background:#f2f2f2}
.ql__chip.is--photo .ql__img{object-fit:cover}
.ql__rule{
  flex:0 0 auto;
  width:1px;
  align-self:stretch;
  background:var(--ql-divider);
  margin-right:8px;      /* the old flex gap, moved to ONE side. See gap:0 above. */
}
.ql__text{
  flex:0 0 auto;
  width:var(--ql-label-w);
  font-size:14px;
  font-weight:400;
  line-height:20px;
  /* ⚠️ DEVIATION-1. Figma sets text-transform:capitalize on this label. Removed,
     deliberately, because it BREAKS BRAND NAMES. Rendered live it turns Tyler's
     "adidas Footwear" into "Adidas Footwear", and across the other 21 locations
     in his sheet it would also produce "Puma", "Adidas Predator" and "Vini Jr.".
     adidas and PUMA are lowercase and uppercase wordmarks respectively and neither
     is a style choice we get to make. Every label in the sheet is already cased
     correctly by hand, so capitalize adds nothing and only ever does damage.
     One line to restore if the design team disagrees. */
  word-break:break-word;
  display:-webkit-box;-webkit-box-orient:vertical;-webkit-line-clamp:3;
  overflow:hidden;                 /* 3 x 20 = 60 < 70. The budget cannot move. */
}

/* text-only variant, for links that have no honest picture: sizes, prices,
   "Turf & Indoor", and for now Lionel Messi. Image box and divider are gone,
   the label sizes to its own text instead of sitting in a 126px column. */
.ql__chip.is--text{padding:4px 8px}   /* no media to bleed, so it keeps Figma's
                                         inset on both sides */
.ql__chip.is--text .ql__text{width:auto;max-width:200px;white-space:nowrap;
  -webkit-line-clamp:1;padding:0 4px}

/* ── ARROWS ───────────────────────────────────────────────────────────────
   Desktop mouse only. display:none by default so a phone never renders them,
   and the height budget is identical either way. They hide at the ends rather
   than sitting there disabled and dead. ── */
.ql__arrow{display:none}

<script>
/* Arrows only. Rule B2 is zero-JS by default and this is the documented
   exception: a mouse wheel cannot scroll a horizontal rail, so desktop needs
   them (design canon). Everything degrades: with JS off you still get a
   scrollable, swipeable, fully linked rail and the arrows simply never wake up.
   Nothing outside the iframe is touched. */
(function () {
  var rail = document.getElementById('ql-rail');
  if (!rail) return;
  var prev = document.querySelector('[data-ql="prev"]');
  var next = document.querySelector('[data-ql="next"]');

  /* Read one chip plus one gap from the DOM rather than hardcoding it, so the
     two breakpoints do not each need their own number. Three chips per click:
     these are much narrower than a product card, and two felt stingy. */
  function step() {
    var chip = rail.querySelector('.ql__item');
    if (!chip) return 300;
    var cs = getComputedStyle(rail);
    var gap = parseFloat(cs.columnGap || cs.gap) || 16;
    return (chip.getBoundingClientRect().width + gap) * 3;
  }
  function update() {
    var max = rail.scrollWidth - rail.clientWidth;
    var atStart = rail.scrollLeft <= 2;
    var atEnd = rail.scrollLeft >= max - 2;
    /* If nothing overflows there is nothing to navigate, so both go away rather
       than sitting there looking live and doing nothing. */
    if (prev) prev.disabled = atStart || max <= 2;
    if (next) next.disabled = atEnd || max <= 2;
  }
  function go(dir) {
    var reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    rail.scrollBy({ left: dir * step(), behavior: reduce ? 'auto' : 'smooth' });
  }
  if (prev) prev.addEventListener('click', function () { go(-1); });
  if (next) next.addEventListener('click', function () { go(1); });

  var raf;
  rail.addEventListener('scroll', function () {
    cancelAnimationFrame(raf); raf = requestAnimationFrame(update);
  }, { passive: true });
  window.addEventListener('resize', update);
  update();
})();
</script>

(function () {
  var BLOCK = "fastr_blocks_quick_links_products_new_v1";
  document.addEventListener("click", function (e) {
    var a = e.target && e.target.closest ? e.target.closest("a[href]") : null;
    if (!a) return;
    try {
      window.parent.postMessage({
        type: "wss-block-click",
        block: BLOCK,
        label: (a.getAttribute("aria-label") || a.textContent || "").replace(/\s+/g, " ").trim().slice(0, 80),
        href: a.href
      }, "*");
    } catch (err) { /* fire-and-forget */ }
  }, true);
})();
</script>
