  // Osmo · Depth Tiles Infinite Loop
  // ⚠️DEVIATION 3: scoped to this page wrapper; section height fixed instead of 100dvh; the loop stays still under reduced motion
  function initDepthTiles() {
    root.querySelectorAll('[data-depth-tiles-init]').forEach(function (container) {
      var list = container.querySelector('[data-depth-tiles-list]');
      var tiles = container.querySelectorAll('[data-depth-tiles-item]');
      var tileCount = tiles.length;
      if (tileCount < 2) return;

      var xMultiplier = 0.65;
      var backScale = 0.5;
      var backOpacity = 1;
      var backDarkness = 1;
      var sideRotateY = 5;
      var perspective = 75;

      var moveDuration = 1.5;
      var startDelay = 0.5;
      var pauseDuration = 0.125;

      var state = { progress: 0 };

      var isActive = false;
      var isHovering = false;
      var hasStarted = false;
      var stepTimeline;
      var delayedCall;
      var startDelayedCall;
      var activeTileIndex = -1;

      gsap.set(list, { perspective: perspective + 'em' });
      gsap.set(tiles, { transformStyle: 'preserve-3d', transformPerspective: perspective * 16 });

      function getRelativeIndex(index) {
        var relative = index - state.progress;
        relative = ((relative + tileCount / 2) % tileCount + tileCount) % tileCount - tileCount / 2;
        return gsap.utils.clamp(-2, 2, relative);
      }

      function getActiveIndex() {
        return ((Math.round(state.progress) % tileCount) + tileCount) % tileCount;
      }

      function updateTileStatus() {
        var currentActiveIndex = getActiveIndex();
        if (currentActiveIndex === activeTileIndex) return;
        activeTileIndex = currentActiveIndex;
        tiles.forEach(function (tile, index) {
          tile.setAttribute('data-depth-tiles-item-status', index === activeTileIndex ? 'active' : 'not-active');
        });
      }

      function renderDepth() {
        var tileWidth = tiles[0].offsetWidth;
        var radiusX = tileWidth * xMultiplier;
        updateTileStatus();
        tiles.forEach(function (tile, index) {
          var relative = getRelativeIndex(index);
          var angle = (relative / 2) * Math.PI;
          var orbitX = Math.sin(angle) * radiusX;
          var orbitDepth = (Math.cos(angle) + 1) / 2;
          var x = relative <= -2 || relative >= 2 ? 0 : orbitX;
          var scale = gsap.utils.interpolate(backScale, 1, orbitDepth);
          var opacity = gsap.utils.interpolate(backOpacity, 1, orbitDepth);
          var brightness = gsap.utils.interpolate(backDarkness, 1, orbitDepth);
          var rotateY = Math.sin(angle) * -sideRotateY;
          var zIndex = Math.round(gsap.utils.interpolate(1, 1000, orbitDepth));
          gsap.set(tile, { x: x, scale: scale, opacity: opacity, rotateY: rotateY, filter: 'brightness(' + brightness + ')', zIndex: zIndex });
        });
      }

      function goToNextTile() {
        if (!isActive || isHovering) return;
        stepTimeline = gsap.timeline({
          paused: true,
          onComplete: function () {
            if (isActive && !isHovering) delayedCall = gsap.delayedCall(pauseDuration, goToNextTile);
          }
        });
        stepTimeline.to(state, { progress: state.progress + 1, duration: moveDuration, ease: 'depth', onUpdate: renderDepth });
        stepTimeline.play();
      }

      function pauseDepth() {
        isActive = false;
        if (stepTimeline) stepTimeline.pause();
        if (delayedCall) delayedCall.pause();
        if (startDelayedCall) startDelayedCall.pause();
      }

      function playDepth() {
        isActive = true;
        if (isHovering) return;
        if (!hasStarted) {
          hasStarted = true;
          startDelayedCall = gsap.delayedCall(startDelay, goToNextTile);
          return;
        }
        if (stepTimeline && stepTimeline.progress() < 1) stepTimeline.play();
        else goToNextTile();
      }

      function handleHoverStart() {
        isHovering = true;
        if (delayedCall) delayedCall.pause();
        if (startDelayedCall) startDelayedCall.pause();
      }

      function handleHoverEnd() {
        isHovering = false;
        if (!isActive) return;
        if (!hasStarted) { playDepth(); return; }
        if (stepTimeline && stepTimeline.progress() < 1) stepTimeline.play();
        else goToNextTile();
      }

      list.addEventListener('pointerover', function (event) {
        if (!event.target.closest('[data-depth-tiles-item]')) return;
        handleHoverStart();
      });
      list.addEventListener('pointerleave', handleHoverEnd);
      list.addEventListener('focusin', handleHoverStart);
      list.addEventListener('focusout', handleHoverEnd);

      renderDepth();
      if (prm) return;

      ScrollTrigger.create({
        trigger: container,
        start: 'top bottom',
        end: 'bottom top',
        onToggle: function (self) { self.isActive ? playDepth() : pauseDepth(); }
      });
    });
  }

  // Osmo · Snowflake Effect
  // ⚠️DEVIATION 4: scoped to the hero; own SVG flake instead of the Osmo image; smaller flakes (scale 0.35 to 0.8); spawning pauses while the hero is off screen; off under reduced motion
  function initSnowflakeEffect() {
    if (prm) return;
    var container = root.querySelector('[data-snowflake-container]');
    if (!container) return;
    if (container.dataset.snowRunning === 'true') return;
    container.dataset.snowRunning = 'true';

    var templates = Array.prototype.slice.call(container.querySelectorAll('[data-snowflake]'));
    if (!templates.length) { container.dataset.snowRunning = 'false'; return; }

    var clamp = function (n, min, max) { return Math.max(min, Math.min(max, n)); };
    var strength = clamp(parseInt(container.dataset.strength || '4', 10) || 0, 0, 10);
    var infinite = (container.dataset.infinite || 'true') !== 'false';

    var durationMin = 8;
    var durationMax = 12;
    var scaleMin = 0.5;
    var scaleMax = 1.2;
    var opacityMin = 0.4;
    var opacityMax = 0.85;

    var spawnRate = gsap.utils.mapRange(0, 10, 0.15, 5.0, strength);
    var maxOnScreen = Math.round(gsap.utils.mapRange(0, 10, 12, 180, strength));
    var burstCount = Math.round(gsap.utils.mapRange(0, 10, 10, 160, strength));

    var running = true;
    var inView = true;
    var activeCount = 0;
    var scheduledCall = null;
    var burstSpawned = 0;

    var getHeight = function () { return container.clientHeight || window.innerHeight; };

    function stop(removeExisting) {
      running = false;
      container.dataset.snowRunning = 'false';
      if (scheduledCall) scheduledCall.kill();
      if (removeExisting !== false) {
        container.querySelectorAll('.snowflake-el.is-spawned').forEach(function (el) { el.remove(); });
        activeCount = 0;
      }
    }

    function cleanupFlake(flake, tweens) {
      tweens.forEach(function (t) { if (t) t.kill(); });
      flake.remove();
      activeCount--;
      if (!infinite && burstSpawned >= burstCount && activeCount <= 0) stop(false);
    }

    function spawnOne() {
      if (!running || !inView) return;
      if (activeCount >= maxOnScreen) return;

      var tpl = templates[Math.floor(Math.random() * templates.length)];
      var flake = tpl.cloneNode(true);
      flake.classList.remove('hidden');
      flake.classList.add('is-spawned');
      flake.style.willChange = 'transform, opacity';

      var scale = gsap.utils.random(scaleMin, scaleMax, 0.001);
      var duration = gsap.utils.random(durationMin, durationMax, 0.001);
      var baseSway = gsap.utils.random(12, 60, 0.1);
      var sway = baseSway * (0.6 + strength / 20);

      var containerWidth = container.clientWidth || window.innerWidth;
      var swayPct = (sway / containerWidth) * 100;
      var padPct = Math.min(20, Math.max(0, swayPct));
      flake.style.left = gsap.utils.random(padPct, 100 - padPct, 0.1) + '%';
      flake.style.opacity = gsap.utils.random(opacityMin, opacityMax, 0.001);

      container.appendChild(flake);
      activeCount++;

      var h = getHeight();
      var startY = -gsap.utils.random(30, Math.min(180, h * 0.25), 1);
      var endY = h + gsap.utils.random(30, Math.min(220, h * 0.35), 1);
      var xStart = gsap.utils.random(-sway, sway, 0.1);
      var xEnd = -xStart;
      var swayDur = gsap.utils.random(1.6, 3.8, 0.001);
      var rotStart = gsap.utils.random(-12, 12, 0.1);
      var rotEnd = gsap.utils.random(-28, 28, 0.1);
      var rotDur = gsap.utils.random(2.2, 5.0, 0.001);

      var fallTween, swayTween, rotTween, fadeTween;
      fallTween = gsap.fromTo(flake,
        { y: startY, xPercent: -50, scale: scale, rotate: rotStart },
        { y: endY, xPercent: -50, ease: 'none', duration: duration,
          onComplete: function () { cleanupFlake(flake, [fallTween, swayTween, rotTween, fadeTween]); } });
      var swayRepeats = Math.max(1, Math.floor(duration / swayDur));
      swayTween = gsap.fromTo(flake, { x: xStart }, { x: xEnd, ease: 'sine.inOut', duration: swayDur, repeat: swayRepeats, yoyo: true });
      var rotRepeats = Math.max(1, Math.floor(duration / rotDur));
      rotTween = gsap.fromTo(flake, { rotate: rotStart }, { rotate: rotEnd, ease: 'sine.inOut', duration: rotDur, repeat: rotRepeats, yoyo: true });
      fadeTween = gsap.to(flake, { opacity: 0, duration: 1, ease: 'power1.out', delay: Math.max(0, duration - 1) });
    }

    function scheduleNext() {
      if (!running) return;
      var avgGap = 1 / spawnRate;
      var nextIn = gsap.utils.random(avgGap * 0.6, avgGap * 1.4, 0.001);
      scheduledCall = gsap.delayedCall(nextIn, function () { spawnOne(); scheduleNext(); });
    }

    if (window.IntersectionObserver) {
      new IntersectionObserver(function (e) { inView = !!(e[0] && e[0].isIntersecting); }, { threshold: 0 }).observe(container);
    }

    if (infinite) {
      var seedCount = Math.round(gsap.utils.mapRange(0, 10, 6, 60, strength));
      for (var i = 0; i < seedCount; i++) gsap.delayedCall(gsap.utils.random(0, 1.2, 0.001), spawnOne);
      scheduleNext();
    } else {
      for (var j = 0; j < burstCount; j++) {
        burstSpawned++;
        gsap.delayedCall(gsap.utils.random(0, 2.0, 0.001), spawnOne);
      }
    }
  }
