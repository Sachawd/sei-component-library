<script>
/* EDITABLE SKIN: ?title= and &key= on the embed URL override the defaults.
   textContent only (no HTML injection), hex validated, 60-char cap.
   No params or no JS: the baked-in defaults above stand. */
(function () {
  try {
    var q = new URLSearchParams(location.search);
    var t = q.get('title');
    if (t) {
      t = t.slice(0, 60);
      document.querySelector('.promo-chip-label').textContent = t;
      var nav = document.querySelector('nav');
      if (nav) nav.setAttribute('aria-label', t + ' categories');
      document.title = 'Intent Selector Chips: ' + t;
    }
    var k = q.get('key');
    if (k) {
      k = k.replace(/^#/, '');
      if (/^([0-9a-fA-F]{3}|[0-9a-fA-F]{6})$/.test(k)) {
        document.documentElement.style.setProperty('--key', '#' + k);
      }
    }
  } catch (e) { /* defaults stand */ }
})();
</script>
<!-- WSS-CLICK-BEACON — attribution-safe click tracking (rules D2). Posts clicks to the
     parent page; pair with Rules and Resources/fastr-parent-click-listener.html. -->
<script>
(function () {
  var BLOCK = "fastr_blocks_intent_selector_chips_sale_v1";
  document.addEventListener("click", function (e) {
    var a = e.target && e.target.closest ? e.target.closest("a[href]") : null;
    if (!a) return;
    try {
      window.parent.postMessage({
        type: "wss-block-click",
        block: BLOCK,
        label: (a.getAttribute("aria-label") || a.textContent || "").replace(/\s+/g, " ").trim().slice(0, 80),
        href: a.href
      }, "*");
    } catch (err) { /* fire-and-forget */ }
  }, true);
})();
</script>
